\usetikzlibrary{positioning, backgrounds}

\definecolor{tilebg}{RGB}{240, 240, 240}
\definecolor{iconcolor}{RGB}{50, 100, 150}

% Helper command to place a tile
% Usage: \tile{column}{row}{icon command}{label}
\newcommand{\tile}[4]{%
	\node[
		fill=tilebg,
		rounded corners=4pt,
		minimum width=1.8cm,
		minimum height=1.3cm,
		align=center,
		draw=gray!50,
		line width=0.5pt,
		font=\sffamily
	] at ({#1*(2cm)}, {-#2*(1.5cm)}) {%
		{\color{iconcolor}\LARGE #3}\\[4pt]
		\scriptsize #4%
	};%
}
