\section{Definitions}

\begin{frame}{FAIR Principles}{What is {\Large \textbf{FAIR}} (data) ?}

	\begin{itemize}
		\item \textbf{Findable}: metadata, permanent links, searchable databases
		\item \textbf{Accessible}: Open and online services, archiving (long retention)
		\item \textbf{Interoperable}: standards and file formats and vocabularies/ontologies
		\item \textbf{Reusable}: Open (source) licenses, (software) context.
	\end{itemize}

	\pause
	\vfill
	\begin{center}
		{\huge \color{red} Demands \textbf{data curation}}
	\end{center}

	\pause
	\vfill
	\begin{block}{  Definition: \textbf{\large Curation}}
		Organizing, cleaning, annotating dataset, ensuring relevance and reusability.\\
	\end{block}
\end{frame}

\begin{frame}[t]{Ontologies}

	\textbf{Ontology (adapted from Wikipedia)}
	\begin{quote}
		an ontology encompasses \textbf{definitions} of the \textbf{categories}, \textbf{properties}, and \textbf{relations} between the \textbf{data entities} of a \textbf{(scientific) topic}.
	\end{quote}
	\pause
	\vfill
	\begin{minipage}{.4\textwidth}
		In practice it is an \textbf{annotated graph} described with the \textit{Resource Description Framework}  (\href{https://www.techtarget.com/searchapparchitecture/definition/Resource-Description-Framework-RDF}{RDF})
		or as a set of \textit{linked classes} (\href{https://pydantic.dev}{Pydantic})
		\newline
		\pause
		\vfill
		e.g. a {\huge \textbf{valid}} mesh file {\huge\textbf{must}} contain nodes and elements.
	\end{minipage}
	\begin{minipage}{.56\textwidth}
		\vspace{-1cm}
		\begin{center}
			\fig{.6}{ontology-exemple}
		\end{center}
	\end{minipage}

	\vfill
	\pause
	\begin{itemize}
		\item \href{https://www.xdmf.org/index.php/XDMF_Model_and_Format}{XDMF} is a XML file (detected by linux) containing \textit{nodes and elements} tags
		\item \textbf{.inp} is the \textbf{Abaqus} input file format, which may, or may not include meshing information
		      \pause
		\item Allows automatic validation/classification $\rightarrow$ Metadata based $\rightarrow$ {\huge \textbf{Fast}!}
		      \pause
		\item \textit{\textbf{Solidipes}} stores metadata with \href{https://rocrate.org}{ROCrate} and validates with \href{https://pydantic.dev}{Pydantic}
	\end{itemize}
	%	Ontology repository: \url{schema.org}
	%	MModa Ontology: \url{https://odahub.io/ontology/}
\end{frame}

\section{Conclusion}

\begin{frame}[t]{Conclusion}{Open Science through Collaborative Data Curation}
	\textbf{Motivation}
	\begin{itemize}
		\item FAIR data (Findable, Accessible, Interoperable, Reusable) requires active \textbf{curation}.
	\end{itemize}
	\vfill
	\pause
	\textbf{Solidipes}, a \textbf{collaborative, plugin-based} curation tool for researchers.
	\begin{itemize}
		\item \textbf{Curation}: \textit{scan $\rightarrow$ load $\rightarrow$ extract metadata $\rightarrow$ validate $\rightarrow$ report $\rightarrow$ publish} (Zenodo, ...)
		\item \textbf{Remote storage} (Rclone, ...)
		\item \textbf{Domain specializations} (plugins, solid mechanics, astrophysics).
		\item \textbf{Ontologies} (Pydantic + RO-Crate) for validation and classification.
	\end{itemize}
	\vfill
	\pause
	\textbf{Next steps}
	\begin{itemize}
		\item \textbf{Workflows/containers} still work toward reproducibility (GUIX, Docker).
		\item \textbf{Reproducibility}: Author recommendations (\href{https://jtcam.github.io/authors-guidelines/authors_guidelines.html}{draft})
		\item \textbf{Reproducibility}: Validations+Ontology $\Rightarrow$ Quality badges
	\end{itemize}
\end{frame}
